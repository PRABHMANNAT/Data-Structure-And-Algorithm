cmake_minimum_required(VERSION 3.20)
project(file_zipper_huffman LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(huffman src/huffman.cpp)
target_include_directories(huffman PUBLIC include)
add_executable(filezip src/main.cpp)
target_link_libraries(filezip PRIVATE huffman)
